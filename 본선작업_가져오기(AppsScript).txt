/**
 * 본선작업 현황(선석별 작업현황 · STS별 작업현황)만 뽑아서 앱에 JSON으로 전달
 * 구글 Apps Script → 웹 앱으로 배포해서 사용합니다.
 * (주소 뒤에 ?type=berth 를 붙이면 선석배정현황 데이터)
 */
const SRC_URL = 'https://snct.sun-kwang.co.kr/infoservice/webpage/vessel/vslWorkStatus.jsp';

function doGet(e) {
  // ?type=berth → 선석배정현황(그래픽), 그 외 → 본선작업 현황
  if (e && e.parameter && e.parameter.type === 'berth') return berth_();
  const cache = CacheService.getScriptCache();
  const hit = cache.get('vsl');
  if (hit) return out_(hit);
  let data;
  try {
    const res = UrlFetchApp.fetch(SRC_URL, {
      muteHttpExceptions: true, followRedirects: true,
      headers: { 'User-Agent': 'Mozilla/5.0' }
    });
    const blob = res.getBlob();
    let html = blob.getDataAsString('UTF-8');
    if (html.indexOf('선석') < 0) html = blob.getDataAsString('EUC-KR'); // 한글 깨짐 방지
    data = {
      ok: res.getResponseCode() === 200 && html.indexOf('선석') >= 0,
      code: res.getResponseCode(),
      at: new Date().toISOString(),
      berths: parseBerths_(html),
      sts: parseSts_(html)
    };
  } catch (err) {
    data = { ok: false, error: String(err), at: new Date().toISOString(), berths: [], sts: [] };
  }
  const json = JSON.stringify(data);
  if (data.ok) cache.put('vsl', json, 50); // 50초 동안은 같은 결과 재사용(회사 사이트 부담 줄이기)
  return out_(json);
}

function out_(s) {
  return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.JSON);
}

function lines_(h) {
  return h.replace(/<svg[\s\S]*?<\/svg>/gi, '')
    .replace(/<br\s*\/?>/gi, '\n')
    .replace(/<\/(div|p|li|tr|span|td|th|b|strong)>/gi, '$&\n')
    .replace(/<[^>]+>/g, '')
    .replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>')
    .split('\n').map(function (x) { return x.replace(/\s+/g, ' ').trim(); }).filter(String);
}

// 선석별 작업현황: 카드마다 제목(선석#1 등) + 안의 글자 줄들
function parseBerths_(html) {
  const a = html.indexOf('선석별 작업현황');
  if (a < 0) return [];
  const b = html.indexOf('STS별 작업현황', a);
  const sec = html.slice(a, b > a ? b : undefined);
  return sec.split(/<div class="vessel-card/).slice(1).map(function (p) {
    const L = lines_('<div ' + p);
    return { title: L[0] || '', empty: /^\s*empty-slot/.test(p), lines: L.slice(1) };
  });
}

// STS별 작업현황: 표의 데이터 줄(칸 글자 배열)
function parseSts_(html) {
  const a = html.indexOf('STS별 작업현황');
  if (a < 0) return [];
  let t = html.slice(a);
  const end = t.indexOf('</table>');
  if (end > 0) t = t.slice(0, end);
  const rows = [];
  t.split(/<tr[^>]*>/i).slice(1).forEach(function (r) {
    if (/<th/i.test(r) || /dash-empty/.test(r)) return;
    const cells = r.split(/<td/i).slice(1).map(function (c) {
      const span = (c.match(/rowspan\s*=\s*"?(\d+)/i) || [])[1];
      const body = c.slice(c.indexOf('>') + 1).split(/<\/td>/i)[0];
      return { t: lines_(body).join(' '), rs: span ? Number(span) : 1 };
    });
    if (cells.length) rows.push(cells);
  });
  return rows;
}

// ===== 선석배정현황(그래픽) =====
const BERTH_URL = 'https://snct.sun-kwang.co.kr/infoservice/webpage/vessel/vslScheduleChart.jsp';

function berth_() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('berth');
  if (hit) return out_(hit);
  let data;
  try {
    const res = UrlFetchApp.fetch(BERTH_URL, { muteHttpExceptions: true, followRedirects: true, headers: { 'User-Agent': 'Mozilla/5.0' } });
    const blob = res.getBlob();
    let html = blob.getDataAsString('UTF-8');
    if (html.indexOf('header_berth') < 0 || html.indexOf('선석') < 0) html = blob.getDataAsString('EUC-KR');
    data = parseBerthChart_(html);
    data.ok = res.getResponseCode() === 200 && data.dates.length > 0;
    data.code = res.getResponseCode();
    data.at = new Date().toISOString();
  } catch (err) {
    data = { ok: false, error: String(err), at: new Date().toISOString(), dates: [], rows: [], blocks: [] };
  }
  const json = JSON.stringify(data);
  if (data.ok) cache.put('berth', json, 120); // 2분 동안 같은 결과 재사용
  return out_(json);
}

function txt_(h) { return lines_(h).join(' '); }

function parseBerthChart_(html) {
  const dates = [];
  html.replace(/class='header_date'[^>]*>([^<]*)</g, function (m, d) { dates.push(d.trim()); return m; });
  const rows = [];
  html.replace(/height='(\d+)'\s+class='header_berth'>([^<]*)</g, function (m, h, name) { rows.push({ name: name.trim(), h: Number(h) }); return m; });
  const parts = html.split(/<div class="vsl-block"/).slice(1);
  const blocks = parts.map(function (p) {
    const st = (p.match(/style="([^"]*)"/) || [])[1] || '';
    const num = function (k) { const m = st.match(new RegExp(k + '\\s*:\\s*([\\d.]+)')); return m ? Number(m[1]) : 0; };
    const bg = (st.match(/background-color\s*:\s*(#[0-9a-fA-F]{3,6})/) || [])[1] || '#888888';
    const boxEnd = p.indexOf('</table>');
    const box = p.slice(0, boxEnd);
    const trs = box.split(/<tr[^>]*>/i).slice(1).map(function (r) {
      return r.split(/<td/i).slice(1).map(function (c) { return txt_(c.slice(c.indexOf('>') + 1).split(/<\/td>/i)[0]); });
    });
    const info = {};
    const infoStart = p.indexOf('class="vsl_info"');
    if (infoStart >= 0) {
      const infoHtml = p.slice(infoStart, p.indexOf('</table>', infoStart));
      infoHtml.split(/<tr[^>]*>/i).slice(1).forEach(function (r) {
        const tds = r.split(/<td/i).slice(1).map(function (c) { return txt_(c.slice(c.indexOf('>') + 1).split(/<\/td>/i)[0]); });
        if (tds.length >= 2) info[tds[0]] = tds[1];
        else if (tds.length === 1 && tds[0]) info['비고'] = tds[0];
      });
    }
    return {
      bg: bg, top: num('top'), left: num('left'), width: num('width'), height: num('height'),
      startNo: (trs[0] || [])[0] || '', endNo: (trs[0] || [])[2] || '',
      route: (trs[1] || [])[0] || '', code: ((trs[2] || [])[0] || '').replace(/\s+/g, ''), bottom: (trs[3] || [])[0] || '',
      info: info
    };
  });
  return { dates: dates, rows: rows, blocks: blocks };
}
